> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship Panopus v1 — a macOS Font Book replacement (activate/deactivate + preview 6,345-font library) as a Studio Manager suite sibling app.

**Architecture:** Tauri v2 fork of StudioManager (React 19 + TS + Tailwind v4 + SQLite). Frontend renders a virtualized grid with lazy `@font-face` injection via the asset protocol.

**Tech Stack:** Tauri v2, React 19, TypeScript, Tailwind v4, tauri-plugin-sql (frontend SQL), rusqlite (Rust-side writes), ttf-parser 0.24, notify 6, core-foundation 0.10, vitest + RTL, cargo test.

**Spec:** `docs/superpowers/specs/2026-09-03-panopus-design.md`

```rust
use std::path::{Path, PathBuf};
use ttf_parser::{name_id, Face};

pub struct FontMeta { pub family: String, pub style: String, pub ps_name: Option<String>,
    pub glyph_count: u16, pub format: String, pub hash: String }

fn name(face: &Face, id: u16) -> Option<String> {
    face.names().into_iter()
        .filter(|n| n.name_id == id && n.is_unicode())
        .find_map(|n| n.to_string())
}

pub fn parse_font(path: &Path) -> Result<FontMeta, String> {
    let data = std::fs::read(path).map_err(|e| e.to_string())?;
    let face = Face::parse(&data, 0).map_err(|e| e.to_string())?;
    let family = name(&face, name_id::TYPOGRAPHIC_FAMILY)
        .or_else(|| name(&face, name_id::FAMILY))
        .ok_or("no family name")?;
    let style = name(&face, name_id::TYPOGRAPHIC_SUBFAMILY)
        .or_else(|| name(&face, name_id::SUBFAMILY)).unwrap_or_else(|| "Regular".into());
    let ext = path.extension().and_then(|e| e.to_str()).unwrap_or("").to_lowercase();
    Ok(FontMeta { family, style, ps_name: name(&face, name_id::POST_SCRIPT_NAME),
        glyph_count: face.number_of_glyphs(),
        format: if ext == "otf" { "otf".into() } else { "ttf".into() },
        hash: format!("{:x}", md5::compute(&data)) })
}

pub fn scan_dir(dir: &Path, is_system: bool) -> Vec<(PathBuf, Result<FontMeta, String>)> {
    walkdir::WalkDir::new(dir).into_iter().filter_map(|e| e.ok())
        .filter(|e| e.file_type().is_file())
        .filter(|e| matches!(e.path().extension().and_then(|x| x.to_str())
            .map(|s| s.to_lowercase()).as_deref(), Some("ttf") | Some("otf")))
        .map(|e| { let p = e.into_path(); let m = parse_font(&p); (p, m) })
        .filter(|(p, m)| !(is_system && (
            p.file_name().and_then(|f| f.to_str()).map_or(false, |f| f.starts_with("Noto"))
            || matches!(m, Ok(meta) if meta.family.starts_with("Noto")))))
        .collect()
}
```

```rust
use core_foundation::{base::TCFType, error::CFErrorRef, url::CFURL};
use std::path::PathBuf;

const K_CT_FONT_MANAGER_SCOPE_USER: u32 = 2; // persistent user scope

extern "C" {
    fn CTFontManagerRegisterFontsForURL(fontURL: core_foundation::url::CFURLRef, scope: u32, error: *mut CFErrorRef) -> bool;
    fn CTFontManagerUnregisterFontsForURL(fontURL: core_foundation::url::CFURLRef, scope: u32, error: *mut CFErrorRef) -> bool;
}

pub fn register_paths(paths: &[PathBuf], register: bool) -> Vec<Result<(), String>> {
    paths.iter().map(|p| {
        let url = CFURL::from_path(p, false).ok_or("bad path")?;
        let mut err: CFErrorRef = std::ptr::null_mut();
        let ok = unsafe { if register {
            CTFontManagerRegisterFontsForURL(url.as_concrete_TypeRef(), K_CT_FONT_MANAGER_SCOPE_USER, &mut err)
        } else {
            CTFontManagerUnregisterFontsForURL(url.as_concrete_TypeRef(), K_CT_FONT_MANAGER_SCOPE_USER, &mut err)
        }};
        // "already registered" (105) / "not registered" (201) count as success for idempotency
        if ok { Ok(()) } else {
            let e = unsafe { core_foundation::error::CFError::wrap_under_create_rule(err) };
            match e.code() { 105 if register => Ok(()), 201 if !register => Ok(()),
                code => Err(format!("CoreText error {code}")) }
        }
    }).collect()
}

pub fn set_active_in_db(conn: &rusqlite::Connection, results: &[(i64, Result<(), String>)], active: bool)
    -> rusqlite::Result<()> {
    let flag = if active { 1 } else { 0 };
    for (id, r) in results { if r.is_ok() {
        conn.execute("UPDATE fonts SET active=?1 WHERE id=?2", rusqlite::params![flag, id])?; } }
    Ok(())
}
```

```rust
use notify::{RecommendedWatcher, RecursiveMode, Watcher};
use std::{sync::mpsc, time::Duration};

pub fn spawn(app: tauri::AppHandle, library: std::path::PathBuf) {
    std::thread::spawn(move || {
        let (tx, rx) = mpsc::channel();
        let mut w: RecommendedWatcher = notify::recommended_watcher(tx).unwrap();
        w.watch(&library, RecursiveMode::Recursive).unwrap();
        loop {
            if rx.recv().is_err() { break; }
            while rx.recv_timeout(Duration::from_secs(2)).is_ok() {} // debounce burst
            let report = crate::indexer::run_full_index(&app); // shared helper from Task 5
            use tauri::Emitter;
            let _ = app.emit("library-changed", &report);
        }
    });
}
```
